\section{Mechanism}
% ============================================================

% Key terms in cards on the left, diagram on the right
\begin{frame}{Mechanism: TLS Migration via \texttt{sendfd}}
  \begin{columns}[T]
    \column{0.38\textwidth}
    \vspace{4pt}
    \begin{tcolorbox}[infocard, title=sendfd]
      \small Pass an open socket fd between processes via a Unix socket\,---\,zero data copy.
    \end{tcolorbox}\medskip
    \begin{tcolorbox}[infocard, title=TLS Context Migration]
      \small Session keys and sequence numbers migrate with the fd; no re-handshake needed.
    \end{tcolorbox}\medskip
    \begin{tcolorbox}[infocard, title=WolfSSL]
      \small Lightweight TLS library used in both gateway and worker container.
    \end{tcolorbox}\medskip
    \begin{tcolorbox}[infocard, title=Zero-overhead handoff]
      \small Worker receives a live, ready-to-read connection directly from the gateway.
    \end{tcolorbox}

    \column{0.62\textwidth}
    \centering\vspace{4pt}
    \includegraphics[width=\linewidth, height=0.84\textheight, keepaspectratio]{images/mechanism.png}
  \end{columns}
\end{frame}

\section{Challenges}
% ============================================================

% Challenge cards on the left, illustration on the right
\begin{frame}{Challenges}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \vspace{4pt}
    \begin{tcolorbox}[challengecard, title=Challenge 1 — HTTP Keep-Alive]
      \small Persistent connections reuse the same TCP socket for multiple requests.
      After migration, how do we handle the subsequent requests on that same connection?
    \end{tcolorbox}\medskip
    \begin{tcolorbox}[challengecard, title=Challenge 2 — TLS Layer]
      \small TLS state (keys, sequence numbers) lives in gateway memory.
      How to move it atomically to the worker without a full re-handshake?
    \end{tcolorbox}\medskip
    \begin{tcolorbox}[challengecard, title=Challenge 3 — Worker Scheduling]
      \small Which container receives the migrated connection?
      Load balancing must remain consistent after the migration.
    \end{tcolorbox}

    \column{0.50\textwidth}
    \centering\vspace{4pt}

        \item Connections: initial (1~req) / keep-alive (50~req)
        \item Protocols: HTTP and HTTPS

\section{Conclusion}
% ============================================================

\begin{frame}{Conclusion}
  \begin{columns}[T]
    \column{0.50\textwidth}
    \begin{tcolorbox}[infocard, title=Summary]
      \begin{itemize}\setlength\itemsep{4pt}\small
        \item TLS migration via \texttt{sendfd} is feasible in a real faasd environment
        \item Prototype achieves lower latency than the classic reverse-proxy approach
        \item Keep-alive support handles persistent connections without any re-handshake
        \item Latency breakdown shows migration cost is small compared to TLS decryption
      \end{itemize}
    \end{tcolorbox}
    \column{0.50\textwidth}
    \begin{tcolorbox}[challengecard, title=Future Work]
      \begin{itemize}\setlength\itemsep{4pt}\small
        \item Extend to multi-node / Kubernetes deployments
        \item Evaluate under higher concurrency
        \item Integrate into upstream faasd / OpenFaaS codebase
        \item Explore QUIC / TLS~1.3 session-ticket migration
      \end{itemize}
    \end{tcolorbox}
  \end{columns}
\end{frame}
